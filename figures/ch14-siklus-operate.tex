\begin{tikzpicture}[

  panah/.style={-{Stealth[length=2mm]}, thick},
  langkah/.style={draw, rounded corners, fill=blue!8, minimum width=2.8cm,

  tekan/.style={draw, rounded corners, fill=orange!12, minimum width=2.8cm,

]
  % Pusat siklus: data tag yang membuat setiap langkah dapat dijalankan.
  \node[draw, rounded corners, fill=praditagreen!15, minimum width=3.4cm,

  \draw[panah] (pantau) -- (anomali);

  \draw[panah] (tindak) -- (tinjau);
  \draw[panah] (tinjau.north) |- (anggaran.west);

  % Data tag dipakai oleh setiap langkah.
  \draw[gray, dashed] (tag) -- (anggaran);

  \draw[gray, dashed] (tag) -- (pantau);
  \draw[gray, dashed] (tag) -- (tinjau);
